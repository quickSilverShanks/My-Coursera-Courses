INSERT INTO invert02(keyword, doc_id)
SELECT DISTINCT s.keyword, id
FROM docs02, unnest(string_to_array(lower(doc), ' ')) s(keyword)
WHERE s.keyword NOT IN (SELECT word FROM stop_words);
